# useMuiForm

The main hook for creating and managing forms with Material-UI components.

## Import

```tsx
import { useMuiForm } from 'usemuiform'
```

## Type Signature

```tsx
function useMuiForm<State extends IState>(
  opts?: UseMuiFormOpts<State>
): UseMuiFormReturn<State>
```

## Parameters

### opts

Configuration object for the form.

```tsx
type UseMuiFormOpts<State> = {
  /** Initial values for the form fields */
  defaultValues?: State

  /** Configuration options for form behavior */
  config?: UseMuiFormConfig
}
```

#### defaultValues

The initial state of your form. Should match your form state type.

```tsx
const form = useMuiForm<{ email: string; name: string }>({
  defaultValues: { email: '', name: '' }
})
```

#### config

Configuration options that override global settings. See [Configuration](/docs/configuration) for details.

```tsx
const form = useMuiForm({
  defaultValues: { email: '' },
  config: {
    requiredFieldErrorMessage: 'Required',
    defaultOpts: {
      required: true,
      lazy: false
    }
  }
})
```

## Return Value

The hook returns an object with the following properties and methods:

### state

The current form state.

```tsx
const { state } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

console.log(state.email) // Current email value
```

### errors

An object containing error messages for each field. `undefined` if no error.

```tsx
const { errors } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

console.log(errors.email) // Error message or undefined
```

### touched

An object tracking which fields have been interacted with.

```tsx
const { touched } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

console.log(touched.email) // true if field has been touched
```

### register

Function to register a field with the form. Returns props to spread onto your input component.

```tsx
function register<Path extends DotPath<State>>(
  name: Path,
  options?: IOptions<PathValue<State, Path>, State>
): RegisterReturn
```

#### name

The field name or dot-notation path for nested fields.

```tsx
register('email')           // Simple field
register('user.email')      // Nested field
register('address.city')    // Nested field
```

#### options

Field-specific configuration.

```tsx
interface IOptions<V, S> {
  /** Whether the field is disabled */
  disabled?: boolean

  /** Custom validation function */
  validate?: ValidateFunc<V, S>

  /** Helper text to display below the field */
  helperText?: string

  /** Whether the field is required */
  required?: boolean

  /** Function to format/transform the value on change */
  format?: FormatFunc<V>

  /** Use lazy (uncontrolled) mode for this field */
  lazy?: boolean
}
```

#### Return value

Returns an object with props to spread onto your MUI component:

```tsx
{
  value: any          // Current field value (for controlled)
  defaultValue: any   // Initial value (for lazy/uncontrolled)
  onChange: function  // Change handler
  onBlur: function    // Blur handler (for lazy inputs)
  error: boolean      // Whether field has an error
  helperText: string  // Error message or custom helper text
  disabled: boolean   // Whether field is disabled
  inputRef: RefObject // Ref for lazy inputs
}
```

### forceValidate

Triggers validation on all fields and returns whether the form is valid.

```tsx
function forceValidate(): boolean
```

Returns `true` if all fields are valid, `false` otherwise.

```tsx
const handleSubmit = () => {
  if (forceValidate()) {
    console.log('Form is valid!', state)
  } else {
    console.log('Form has errors')
  }
}
```

### clear

Resets the form to its default values.

```tsx
function clear(): void
```

```tsx
const handleReset = () => {
  clear()
}
```

### setValue

Programmatically set the value of a field.

```tsx
function setValue<Path extends DotPath<State>>(
  name: Path,
  value: PathValue<State, Path>
): void
```

```tsx
const { setValue } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

setValue('email', 'user@example.com')
```

### isValid

Boolean indicating whether the form is currently valid.

```tsx
const { isValid } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

console.log(isValid) // true or false
```

### isAnyTouched

Boolean indicating whether any field has been touched.

```tsx
const { isAnyTouched } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

console.log(isAnyTouched) // true if any field touched
```

### isChanged

Boolean indicating whether the form state differs from default values.

```tsx
const { isChanged } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

console.log(isChanged) // true if form has changes
```

## Complete Example

```tsx
import { useMuiForm, type ValidateFunc } from 'usemuiform'
import { TextField, Button } from '@mui/material'

type FormState = {
  email: string
  name: string
}

export default function MyForm() {
  const {
    state,
    errors,
    touched,
    register,
    forceValidate,
    clear,
    setValue,
    isValid,
    isAnyTouched,
    isChanged
  } = useMuiForm<FormState>({
    defaultValues: { email: '', name: '' },
    config: {
      requiredFieldErrorMessage: 'This field is required'
    }
  })

  const emailValidator: ValidateFunc<string, FormState> = (value) => {
    if (!value.includes('@')) return 'Invalid email'
    return true
  }

  const handleSubmit = () => {
    if (forceValidate()) {
      console.log('Submitted:', state)
      clear()
    }
  }

  const handlePrefill = () => {
    setValue('email', 'test@example.com')
    setValue('name', 'John Doe')
  }

  return (
    <div>
      <TextField
        label="Email"
        {...register('email', {
          required: true,
          validate: emailValidator
        })}
        fullWidth
      />

      <TextField
        label="Name"
        {...register('name', { required: true })}
        fullWidth
      />

      <div>
        <p>Is Valid: {isValid ? 'Yes' : 'No'}</p>
        <p>Any Touched: {isAnyTouched ? 'Yes' : 'No'}</p>
        <p>Has Changes: {isChanged ? 'Yes' : 'No'}</p>
      </div>

      <Button onClick={handleSubmit} disabled={!isValid}>
        Submit
      </Button>
      <Button onClick={handlePrefill}>
        Prefill
      </Button>
      <Button onClick={clear}>
        Clear
      </Button>
    </div>
  )
}
```
