# UseMuiFormConfigProvider

A React context provider for setting global configuration defaults for all forms in your application.

## Import

```tsx
import { UseMuiFormConfigProvider } from 'usemuiform'
```

## Type Signature

```tsx
function UseMuiFormConfigProvider({
  config,
  children
}: {
  config: UseMuiFormConfig
  children: ReactNode
}): JSX.Element
```

## Props

### config

Configuration object to apply globally.

```tsx
interface UseMuiFormConfig {
  /** Custom error message for required fields */
  requiredFieldErrorMessage?: string

  /** Default options for all fields */
  defaultOpts?: {
    /** Whether fields are required by default */
    required?: boolean
    /** Whether fields use lazy (uncontrolled) mode by default */
    lazy?: boolean
  }
}
```

### children

React components that should have access to the configuration.

## Usage

Wrap your application (or part of it) with the provider:

```tsx
import { UseMuiFormConfigProvider } from 'usemuiform'

function App() {
  return (
    <UseMuiFormConfigProvider config={{
      requiredFieldErrorMessage: 'This field is required!',
      defaultOpts: {
        required: false,
        lazy: true
      }
    }}>
      <YourApp />
    </UseMuiFormConfigProvider>
  )
}
```

## Configuration Options

### requiredFieldErrorMessage

Customize the error message displayed for required fields.

```tsx
<UseMuiFormConfigProvider config={{
  requiredFieldErrorMessage: 'Please fill out this field'
}}>
  <App />
</UseMuiFormConfigProvider>
```

Default: `"Field is required"`

### defaultOpts.required

Set whether fields are required by default.

```tsx
<UseMuiFormConfigProvider config={{
  defaultOpts: {
    required: false  // All fields optional by default
  }
}}>
  <App />
</UseMuiFormConfigProvider>
```

Default: `true`

When set to `false`, you can opt-in to required validation per field:

```tsx
<TextField {...register('email', { required: true })} />
```

### defaultOpts.lazy

Set whether fields use lazy (uncontrolled) mode by default.

```tsx
<UseMuiFormConfigProvider config={{
  defaultOpts: {
    lazy: true  // All fields use lazy mode by default
  }
}}>
  <App />
</UseMuiFormConfigProvider>
```

Default: `false`

When set to `true`, fields only update state on blur instead of every keystroke.

## Nesting Providers

You can nest providers to create different configuration zones:

```tsx
function App() {
  return (
    <UseMuiFormConfigProvider config={{
      defaultOpts: { required: false }
    }}>
      <OuterForms />

      <UseMuiFormConfigProvider config={{
        defaultOpts: { required: true, lazy: true }
      }}>
        <InnerForms />
      </UseMuiFormConfigProvider>
    </UseMuiFormConfigProvider>
  )
}
```

Inner configurations override outer ones.

## Complete Example

```tsx
import {
  UseMuiFormConfigProvider,
  useMuiForm,
  type UseMuiFormConfig
} from 'usemuiform'
import { TextField, Button } from '@mui/material'

// Define your global config
const globalConfig: UseMuiFormConfig = {
  requiredFieldErrorMessage: 'Please complete this field',
  defaultOpts: {
    required: false,  // Fields optional by default
    lazy: true        // Use lazy mode by default
  }
}

export function App() {
  return (
    <UseMuiFormConfigProvider config={globalConfig}>
      <MainApp />
    </UseMuiFormConfigProvider>
  )
}

type FormState = {
  email: string
  name: string
  newsletter: boolean
}

function MainApp() {
  // This form inherits the global config
  const { register, forceValidate } = useMuiForm<FormState>({
    defaultValues: { email: '', name: '', newsletter: false }
  })

  const handleSubmit = () => {
    if (forceValidate()) {
      console.log('Valid!')
    }
  }

  return (
    <div>
      {/* Optional and lazy (from global config) */}
      <TextField
        label="Name"
        {...register('name')}
      />

      {/* Required (field override), lazy (from config) */}
      <TextField
        label="Email"
        {...register('email', { required: true })}
      />

      {/* Optional and controlled (overrides lazy config) */}
      <TextField
        label="Newsletter"
        {...register('newsletter', { lazy: false })}
      />

      <Button onClick={handleSubmit}>Submit</Button>
    </div>
  )
}
```

## TypeScript

Full type safety for configuration:

```tsx
import type { UseMuiFormConfig } from 'usemuiform'

// Type-checked config
const config: UseMuiFormConfig = {
  requiredFieldErrorMessage: 'Required',
  defaultOpts: {
    required: true,
    lazy: false
  }
}

// TypeScript will catch errors
const invalidConfig: UseMuiFormConfig = {
  requiredFieldErrorMessage: 123,  // ❌ Error: should be string
  defaultOpts: {
    required: 'yes'  // ❌ Error: should be boolean
  }
}
```
