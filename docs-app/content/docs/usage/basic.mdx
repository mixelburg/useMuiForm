# Using with MUI Components

useMuiForm's `register` function automatically provides the correct props for MUI components. Here's how to use it with common MUI inputs.

## TextField

The most common use case - works with all TextField variants:

```tsx
import { useMuiForm } from 'usemuiform'
import { TextField } from '@mui/material'

const { register } = useMuiForm<{ email: string }>({
  defaultValues: { email: '' }
})

<TextField
  label="Email"
  type="email"
  {...register('email', {
    required: 'Email is required',
    pattern: {
      value: /^[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}$/i,
      message: 'Invalid email address'
    }
  })}
/>
```

## Select (with TextField)

Use TextField with `select` prop:

```tsx
import { TextField, MenuItem } from '@mui/material'

<TextField
  select
  label="Role"
  {...register('role')}
>
  <MenuItem value="admin">Admin</MenuItem>
  <MenuItem value="user">User</MenuItem>
  <MenuItem value="guest">Guest</MenuItem>
</TextField>
```

## Select (standalone)

Or use the standalone Select component:

```tsx
import { Select, MenuItem } from '@mui/material'

<Select
  {...register('role')}
>
  <MenuItem value="admin">Admin</MenuItem>
  <MenuItem value="user">User</MenuItem>
  <MenuItem value="guest">Guest</MenuItem>
</Select>
```

## Multiple Select

For multi-select, the value will be an array:

```tsx
type FormState = {
  roles: string[]
}

<Select
  multiple
  {...register('roles')}
>
  <MenuItem value="admin">Admin</MenuItem>
  <MenuItem value="user">User</MenuItem>
  <MenuItem value="guest">Guest</MenuItem>
</Select>
```

## Checkbox

Checkboxes need special handling - extract `helperText` and `error` separately:

```tsx
import { FormControlLabel, Checkbox, FormHelperText, FormGroup } from '@mui/material'

<FormGroup>
  {(() => {
    const { helperText, error, ...checkboxProps } = register('isActive')
    return (
      <>
        <FormControlLabel
          label="Is Active"
          control={<Checkbox {...checkboxProps} />}
        />
        <FormHelperText error={error}>{helperText}</FormHelperText>
      </>
    )
  })()}
</FormGroup>
```

Or simpler without helper text:

```tsx
<FormControlLabel
  label="Accept Terms"
  control={<Checkbox {...register('acceptTerms')} />}
/>
```

## Switch

Same pattern as Checkbox:

```tsx
import { FormControlLabel, Switch } from '@mui/material'

<FormControlLabel
  label="Enable Notifications"
  control={<Switch {...register('notifications')} />}
/>
```

## DatePicker (MUI X)

Works with MUI X date pickers:

```tsx
import { DatePicker } from '@mui/x-date-pickers'
import { LocalizationProvider } from '@mui/x-date-pickers'
import { AdapterDayjs } from '@mui/x-date-pickers/AdapterDayjs'
import dayjs from 'dayjs'

type FormState = {
  birthDate: dayjs.Dayjs
}

const { register } = useMuiForm<FormState>({
  defaultValues: { birthDate: dayjs() }
})

<LocalizationProvider dateAdapter={AdapterDayjs}>
  <DatePicker
    label="Birth Date"
    {...register('birthDate', {
      validate: (value) => {
        if (value.isBefore(dayjs().year(2000))) {
          return 'Date must be after 2000'
        }
        return true
      }
    })}
  />
</LocalizationProvider>
```

## DateTimePicker

Same as DatePicker:

```tsx
import { DateTimeField } from '@mui/x-date-pickers'

<DateTimeField
  label="Appointment"
  {...register('appointmentTime')}
/>
```

## Nested Fields

Use dot notation for nested objects:

```tsx
type FormState = {
  person: {
    name: string
    email: string
  }
}

<TextField
  label="Name"
  {...register('person.name', { required: 'Name is required' })}
/>

<TextField
  label="Email"
  {...register('person.email', { required: 'Email is required' })}
/>
```

## Complete Example

```tsx
import { useMuiForm } from 'usemuiform'
import {
  TextField,
  Button,
  MenuItem,
  FormControlLabel,
  Checkbox,
  FormGroup,
  FormHelperText,
  Stack
} from '@mui/material'

type FormState = {
  email: string
  role: 'admin' | 'user' | 'guest'
  isActive: boolean
}

export default function MyForm() {
  const { register, handleSubmit } = useMuiForm<FormState>({
    defaultValues: {
      email: '',
      role: 'user',
      isActive: false
    }
  })

  const onSubmit = (data: FormState) => {
    console.log('Form data:', data)
  }

  return (
    <Stack spacing={2}>
      <TextField
        label="Email"
        type="email"
        {...register('email', {
          required: 'Email is required',
          pattern: {
            value: /^[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}$/i,
            message: 'Invalid email'
          }
        })}
      />

      <TextField
        select
        label="Role"
        {...register('role')}
      >
        <MenuItem value="admin">Admin</MenuItem>
        <MenuItem value="user">User</MenuItem>
        <MenuItem value="guest">Guest</MenuItem>
      </TextField>

      <FormGroup>
        {(() => {
          const { helperText, error, ...checkboxProps } = register('isActive')
          return (
            <>
              <FormControlLabel
                label="Is Active"
                control={<Checkbox {...checkboxProps} />}
              />
              <FormHelperText error={error}>{helperText}</FormHelperText>
            </>
          )
        })()}
      </FormGroup>

      <Button variant="contained" onClick={handleSubmit(onSubmit)}>
        Submit
      </Button>
    </Stack>
  )
}
```
