# Nested Fields

useMuiForm supports nested object structures using dot notation for field paths, powered by lodash's `get` and `set` functions.

## Defining Nested State

Create a form state type with nested objects:

```tsx
type FormState = {
  user: {
    name: string
    email: string
  }
  address: {
    street: string
    city: string
    zipCode: string
  }
}
```

## Registering Nested Fields

Use dot notation to reference nested fields:

```tsx
const { register } = useMuiForm<FormState>({
  defaultValues: {
    user: { name: '', email: '' },
    address: { street: '', city: '', zipCode: '' }
  }
})

// Register nested fields
<TextField
  label="Name"
  {...register('user.name', { required: true })}
/>

<TextField
  label="Email"
  {...register('user.email', { required: true })}
/>

<TextField
  label="Street"
  {...register('address.street', { required: true })}
/>
```

## Type Safety

TypeScript provides full autocomplete and type checking for nested paths:

```tsx
// ✅ Valid - TypeScript knows these paths exist
register('user.name')
register('address.city')

// ❌ Error - TypeScript catches invalid paths
register('user.age')        // Property 'age' does not exist
register('address.country') // Property 'country' does not exist
```

## Accessing Nested Values

Access nested state values directly:

```tsx
const { state } = useMuiForm<FormState>({
  defaultValues: {
    user: { name: '', email: '' },
    address: { street: '', city: '', zipCode: '' }
  }
})

console.log(state.user.name)
console.log(state.address.city)
```

## Complete Example

```tsx
import { useMuiForm } from 'usemuiform'
import { TextField, Button, Typography } from '@mui/material'

type FormState = {
  user: {
    firstName: string
    lastName: string
    email: string
  }
  address: {
    street: string
    city: string
    state: string
    zipCode: string
  }
}

export default function NestedForm() {
  const { state, register, forceValidate } = useMuiForm<FormState>({
    defaultValues: {
      user: { firstName: '', lastName: '', email: '' },
      address: { street: '', city: '', state: '', zipCode: '' }
    }
  })

  const handleSubmit = () => {
    if (forceValidate()) {
      console.log('Form data:', state)
    }
  }

  return (
    <div>
      <Typography variant="h6">User Information</Typography>
      <TextField
        label="First Name"
        {...register('user.firstName', { required: true })}
        fullWidth
      />
      <TextField
        label="Last Name"
        {...register('user.lastName', { required: true })}
        fullWidth
      />
      <TextField
        label="Email"
        type="email"
        {...register('user.email', { required: true })}
        fullWidth
      />

      <Typography variant="h6" sx={{ mt: 3 }}>Address</Typography>
      <TextField
        label="Street"
        {...register('address.street', { required: true })}
        fullWidth
      />
      <TextField
        label="City"
        {...register('address.city', { required: true })}
        fullWidth
      />
      <TextField
        label="State"
        {...register('address.state', { required: true })}
        fullWidth
      />
      <TextField
        label="ZIP Code"
        {...register('address.zipCode', { required: true })}
        fullWidth
      />

      <Button variant="contained" onClick={handleSubmit}>
        Submit
      </Button>
    </div>
  )
}
```

## Validation with Nested Fields

Validation works the same way with nested fields:

```tsx
import type { ValidateFunc } from 'usemuiform'

const emailValidator: ValidateFunc<string, FormState> = (value) => {
  if (!value.includes('@')) return 'Invalid email'
  return true
}

<TextField
  label="Email"
  {...register('user.email', {
    required: true,
    validate: emailValidator
  })}
/>
```

## Cross-Field Validation with Nested State

Access other nested fields during validation:

```tsx
const zipCodeValidator: ValidateFunc<string, FormState> = (value, state) => {
  // Access other nested fields
  if (state.address.state === 'CA' && !value.startsWith('9')) {
    return 'California ZIP codes should start with 9'
  }
  return true
}

<TextField
  label="ZIP Code"
  {...register('address.zipCode', {
    required: true,
    validate: zipCodeValidator
  })}
/>
```
