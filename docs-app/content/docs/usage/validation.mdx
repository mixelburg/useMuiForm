# Validation

useMuiForm provides flexible validation options for your form fields, including built-in required field validation and custom validation functions.

## Required Fields

Mark a field as required by passing `required: true` in the register options:

```tsx
<TextField
  label="Email"
  {...register('email', { required: true })}
/>
```

When a required field is empty, it will display an error message.

## Custom Validation

Define custom validation logic using the `validate` option:

```tsx
import type { ValidateFunc } from 'usemuiform'

const emailValidator: ValidateFunc<string, FormState> = (value, state) => {
  if (value.length < 5) return 'Email must be at least 5 characters long'
  if (!value.includes('@')) return 'Email must contain @'
  return true
}

<TextField
  label="Email"
  {...register('email', {
    required: true,
    validate: emailValidator
  })}
/>
```

The validate function receives:
- `value` - The current field value
- `state` - The entire form state (useful for cross-field validation)

Return `true` if valid, or a string error message if invalid.

## Cross-Field Validation

Access other field values through the `state` parameter:

```tsx
const passwordConfirmValidator: ValidateFunc<string, FormState> = (value, state) => {
  if (value !== state.password) {
    return 'Passwords do not match'
  }
  return true
}

<TextField
  label="Confirm Password"
  type="password"
  {...register('confirmPassword', {
    required: true,
    validate: passwordConfirmValidator
  })}
/>
```

## Validation Timing

By default, validation occurs:
- On blur (when field loses focus)
- On change (after the field has been touched)
- When `forceValidate()` is called

## Force Validation

Trigger validation on all fields programmatically:

```tsx
const handleSubmit = () => {
  if (forceValidate()) {
    // All fields are valid
    console.log('Form is valid!', state)
  } else {
    // Some fields have errors
    console.log('Form has errors')
  }
}
```

## Helper Text

Provide static helper text that appears when there's no error:

```tsx
<TextField
  label="Email"
  {...register('email', {
    required: true,
    helperText: 'We will never share your email'
  })}
/>
```

The helper text is automatically replaced by error messages when validation fails.

## Custom Error Messages

Customize the required field error message globally via configuration:

```tsx
import { UseMuiFormConfigProvider } from 'usemuiform'

<UseMuiFormConfigProvider config={{
  requiredFieldErrorMessage: 'This field is mandatory'
}}>
  <App />
</UseMuiFormConfigProvider>
```

Or per-form:

```tsx
const form = useMuiForm({
  defaultValues: { email: '' },
  config: {
    requiredFieldErrorMessage: 'Please fill this field'
  }
})
```

## Error State Access

Access validation errors directly:

```tsx
const { errors } = useMuiForm<FormState>({
  defaultValues: { email: '' }
})

// errors.email will contain the error message or undefined
console.log(errors.email)
```

## Complete Example

```tsx
import { useMuiForm, type ValidateFunc } from 'usemuiform'
import { TextField, Button } from '@mui/material'

type FormState = {
  email: string
  password: string
  confirmPassword: string
}

export default function ValidationExample() {
  const { state, register, forceValidate, errors } = useMuiForm<FormState>({
    defaultValues: { email: '', password: '', confirmPassword: '' }
  })

  const emailValidator: ValidateFunc<string, FormState> = (value) => {
    if (value.length < 5) return 'Email must be at least 5 characters'
    if (!value.includes('@')) return 'Email must contain @'
    return true
  }

  const passwordValidator: ValidateFunc<string, FormState> = (value) => {
    if (value.length < 8) return 'Password must be at least 8 characters'
    if (!/[A-Z]/.test(value)) return 'Password must contain uppercase letter'
    return true
  }

  const confirmValidator: ValidateFunc<string, FormState> = (value, state) => {
    if (value !== state.password) return 'Passwords do not match'
    return true
  }

  const handleSubmit = () => {
    if (forceValidate()) {
      console.log('Form submitted:', state)
    }
  }

  return (
    <div>
      <TextField
        label="Email"
        {...register('email', {
          required: true,
          validate: emailValidator,
          helperText: 'Enter your email address'
        })}
        fullWidth
      />

      <TextField
        label="Password"
        type="password"
        {...register('password', {
          required: true,
          validate: passwordValidator
        })}
        fullWidth
      />

      <TextField
        label="Confirm Password"
        type="password"
        {...register('confirmPassword', {
          required: true,
          validate: confirmValidator
        })}
        fullWidth
      />

      <Button variant="contained" onClick={handleSubmit}>
        Submit
      </Button>
    </div>
  )
}
```
